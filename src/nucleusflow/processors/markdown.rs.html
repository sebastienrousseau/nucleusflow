1//! # Markdown Processing Module
2//!
3//! Provides a secure and flexible Markdown processing framework with configurable options,
4//! metadata extraction, and content sanitization. This module focuses on safety,
5//! performance, and extensibility.
6//!
7//! ## Key Features
8//!
9//! - **Safe Processing**: Secure content handling with robust sanitization
10//! - **Metadata Extraction**: YAML frontmatter parsing with type-safe handling
11//! - **Table of Contents**: Automatic generation of nested TOC structures
12//! - **Configurable Options**: Support for tables, footnotes, and strikethrough
13//! - **Content Validation**: Protection against XSS and other injection attacks
14//!
15//! ## Example Usage
16//!
17//! ```rust
18//! use nucleusflow::processors::markdown::MarkdownProcessor;
19//! use nucleusflow::core::traits::Processor;
20//!
21//! let processor = MarkdownProcessor::new()
22//!     .with_tables(true)
23//!     .with_footnotes(true)
24//!     .with_strikethrough(true);
25//!
26//! let content = r#"---
27//! title: My Post
28//! tags: [rust, markdown]
29//! ---
30//! # Hello World
31//!
32//! This is a **markdown** document.
33//! "#;
34//!
35//! let result = processor.process(content.to_string(), None).unwrap();
36//! ```

85/// Configuration options for markdown processing.

87pub struct ProcessorConfig {

147/// Processor for Markdown content with advanced features and security.
148#[derive(Debug, Clone)]
149pub struct MarkdownProcessor {
150    options: MarkdownOptions,
151    config: ProcessorConfig,
152    /// Cache of allowed HTML tags for faster sanitization
153    allowed_tags: Arc<HashSet<String>>,

156impl MarkdownProcessor {
157    /// Creates a new MarkdownProcessor with default settings.
158    pub fn new() -> Self {
159        let allowed_tags = ALLOWED_HTML_TAGS
160            .iter()
161            .map(|&tag| tag.to_string())
162            .collect();
163
164        Self {
165            options: MarkdownOptions::empty(),
166            config: ProcessorConfig::default(),
167            allowed_tags: Arc::new(allowed_tags),
168        }

207    /// Extracts and validates metadata from Markdown content.
208    fn extract_metadata(
209        &self,
210        content: &str,
211    ) -> Result<ContentMetadata> {
212        let mut metadata = ContentMetadata::default();
213        let mut lines = content.lines();
214
215        // Handle YAML frontmatter
216        if content.starts_with("---\n") {
217            let mut frontmatter = String::with_capacity(1024);
218            let _ = lines.next(); // Skip first "---"
219
220            for line in lines.by_ref() {
221                if line == "---" {
222                    break;
223                }
224                frontmatter.push_str(line);
225                frontmatter.push('\n');
226            }
227
228            if let Ok(yaml) =
229                from_str::<HashMap<String, JsonValue>>(&frontmatter)
230            {
231                Self::process_metadata(&mut metadata, yaml)?;
232            }
233        }
234
235        // Extract title from first H1 if not found in frontmatter
236        if metadata.title.is_none() {
237            for line in content.lines() {
238                if let Some(title) = line.strip_prefix("# ") {
239                    metadata.title = Some(title.trim().to_string());
240                    break;
241                }
242            }
243        }
244
245        Ok(metadata)
246    }
247
248    /// Processes and validates metadata from YAML frontmatter.
249    fn process_metadata(
250        metadata: &mut ContentMetadata,
251        yaml: HashMap<String, JsonValue>,
252    ) -> Result<()> {
253        for (key, value) in yaml {
254            match key.as_str() {
255                "title" => {
256                    metadata.title = value
257                        .as_str()
258                        .map(|s| s.trim().to_string())
259                        .filter(|s| !s.is_empty());
260                }
261                "description" => {
262                    metadata.description = value
263                        .as_str()
264                        .map(|s| s.trim().to_string())
265                        .filter(|s| !s.is_empty());
266                }
267                "date" => {
268                    metadata.date = value
269                        .as_str()
270                        .map(|s| s.trim().to_string())
271                        .filter(|s| !s.is_empty());
272                }
273                "tags" => {
274                    if let Some(tags) = value.as_array() {
275                        metadata.tags = tags
276                            .iter()
277                            .filter_map(|v| {
278                                v.as_str()
279                                    .map(|s| s.trim().to_string())
280                                    .filter(|s| !s.is_empty())
281                            })
282                            .collect();
283                    }
284                }
285                _ => {
286                    let _ = metadata.custom.insert(key, value);
287                }
288            }
289        }
290
291        Ok(())
292    }

294    /// Generates an accessible Table of Contents.
295    fn generate_toc(&self, content: &str) -> Result<String> {
296        let mut toc = String::from(
297        "<nav class=\"toc\" aria-label=\"Table of Contents\">\n<ul>\n",
298    );
299        let mut entries = Vec::new();
300        let parser = Parser::new_ext(content, self.options);
301        let mut current_text = String::new();
302        let mut current_level = None;
303
304        for event in parser {
305            match event {
306                Event::Start(Tag::Heading { level, .. }) => {
307                    current_text.clear();
308                    current_level = Some(level);
309                }
310                Event::Text(text) => {
311                    current_text.push_str(&text);
312                }
313                Event::End(_) => {
314                    if let Some(level) = current_level.take() {
315                        let level_num = match level {
316                            HeadingLevel::H1 => 1,
317                            HeadingLevel::H2 => 2,
318                            HeadingLevel::H3 => 3,
319                            HeadingLevel::H4 => 4,
320                            HeadingLevel::H5 => 5,
321                            HeadingLevel::H6 => 6,
322                        };
323
324                        if level_num <= self.config.toc_max_level {
325                            let id =
326                                self.generate_heading_id(&current_text);
327                            entries.push(TocEntry {
328                                text: current_text.clone(),
329                                level: level_num,
330                                id,
331                            });
332                        }
333                    }
334                }
335                _ => {}
336            }
337        }
338
339        self.build_toc_html(&mut toc, &entries)?;
340        toc.push_str("</ul>\n</nav>");
341        Ok(toc)
342    }
343
344    /// Generates a unique ID for a heading.
345    fn generate_heading_id(&self, text: &str) -> String {
346        text.to_lowercase()
347            .chars()
348            .filter_map(|c| match c {
349                'a'..='z' | '0'..='9' => Some(c),
350                ' ' | '-' | '_' => Some('-'),
351                _ => None,
352            })
353            .collect()
354    }
355
356    /// Builds the HTML structure for the Table of Contents.
357    fn build_toc_html(
358        &self,
359        toc: &mut String,
360        entries: &[TocEntry],
361    ) -> Result<()> {
362        let mut current_level = 1;
363
364        for entry in entries {
365            while entry.level > current_level {
366                toc.push_str("<ul>\n");
367                current_level += 1;
368            }
369            while entry.level < current_level {
370                toc.push_str("</ul>\n");
371                current_level -= 1;
372            }
373
374            toc.push_str(&format!(
375                "<li><a href=\"#{}\" aria-label=\"{}\">{}</a></li>\n",
376                entry.id, entry.text, entry.text
377            ));
378        }
379
380        while current_level > 1 {
381            toc.push_str("</ul>\n");
382            current_level -= 1;
383        }
384
385        Ok(())
386    }

388    /// Sanitizes HTML content to prevent XSS and other injection attacks.
389    fn sanitize_html(&self, html: &str) -> Result<String> {
390        let mut output = String::with_capacity(html.len());
391        let mut in_tag = false;
392        let mut current_tag = String::new();
393
394        for c in html.chars() {
395            match c {
396                '<' => {
397                    in_tag = true;
398                    current_tag.clear();
399                }
400                '>' if in_tag => {
401                    in_tag = false;
402                    let tag_name = current_tag
403                        .split_whitespace()
404                        .next()
405                        .unwrap_or("")
406                        .trim_start_matches('/')
407                        .to_lowercase();
408
409                    if self.allowed_tags.contains(&tag_name) {
410                        output.push('<');
411                        output.push_str(&current_tag);
412                        output.push('>');
413                    }
414                }
415                _ if in_tag => {
416                    current_tag.push(c);
417                }
418                _ => {
419                    if !in_tag {
420                        output.push(c);
421                    }
422                }
423            }
424        }
425
426        Ok(output)
427    }

429    /// Validates that the content is safe to process.
430    fn validate(&self, content: &str) -> Result<()> {
431        // Check content size
432        if content.len() > MAX_CONTENT_SIZE {
433            return Err(ProcessingError::ContentProcessing {
434                details: format!(
435                    "Content exceeds maximum size of {} bytes",
436                    MAX_CONTENT_SIZE
437                ),
438                source: None,
439            });
440        }
441
442        // Check for empty content
443        if content.trim().is_empty() {
444            return Err(ProcessingError::ContentProcessing {
445                details: "Content cannot be empty".to_string(),
446                source: None,
447            });
448        }
449
450        // Check for suspicious patterns
451        let suspicious_patterns = [
452            "javascript:",
453            "data:",
454            "vbscript:",
455            "onclick",
456            "onerror",
457            "onload",
458            "eval(",
459        ];
460
461        for pattern in &suspicious_patterns {
462            if content.to_lowercase().contains(pattern) {
463                return Err(ProcessingError::ContentProcessing {
464                    details: format!(
465                        "Suspicious content pattern detected: {}",
466                        pattern
467                    ),
468                    source: None,
469                });
470            }
471        }
472
473        Ok(())
474    }

483impl Processor for MarkdownProcessor {
484    type Input = String;
485    type Output = String;
486    type Context = JsonValue;
487
488    fn process(
489        &self,
490        content: String,
491        context: Option<&Self::Context>,
492    ) -> Result<Self::Output> {
493        // Validate content
494        self.validate(&content)?;
495
496        // Extract metadata
497        let metadata = self.extract_metadata(&content)?;
498
499        // Parse configuration from context
500        let config: ProcessorConfig = context
501            .and_then(|ctx| serde_json::from_value(ctx.clone()).ok())
502            .unwrap_or_default();
503
504        // Parse Markdown to HTML
505        let parser = Parser::new_ext(&content, self.options);
506        let mut html_output = String::with_capacity(content.len() * 2);
507        html::push_html(&mut html_output, parser);
508
509        // Generate and prepend TOC if enabled
510        if config.toc {
511            let toc = self.generate_toc(&content)?;
512            println!("Generated ToC: {}", toc); // Debugging line
513            html_output = format!("{}\n{}", toc, html_output);
514        }
515
516        // Sanitize if enabled
517        let processed = if config.sanitize {
518            self.sanitize_html(&html_output)?
519        } else {
520            html_output
521        };
522
523        // Add metadata as JSON-LD if present
524        if !metadata.custom.is_empty() {
525            let json_ld = serde_json::to_string(&metadata.custom)
526                .map_err(|e| ProcessingError::ContentProcessing {
527                    details: "Failed to serialize metadata".to_string(),
528                    source: Some(Box::new(e)),
529                })?;
530            Ok(format!(
531                "{}\n<script type=\"application/ld+json\">{}</script>",
532                processed, json_ld
533            ))
534        } else {
535            Ok(processed)
536        }
537    }
538}
