1//! # Content Processors Module
2//!
3//! This module provides various content processors for different formats and types.
4//! Each processor implements the core processing traits and handles specific content
5//! types (e.g., Markdown, HTML, etc.).
6//!
7//! ## Available Processors
8//!
9//! - [`markdown`]: Processes Markdown content with support for frontmatter and extensions
10//!
11//! ## Usage
12//!
13//! ```rust,no_run
14//! use nucleusflow::processors::markdown::MarkdownProcessor;
15//! use nucleusflow::core::traits::Processor;
16//! use nucleusflow::ContentProcessor;
17//!
18//! let processor = MarkdownProcessor::new()
19//!     .with_tables(true)
20//!     .with_footnotes(true);
21//!
22//! let content = "# Hello World\n\nThis is a test.";
23//! let result = processor.process(content.to_string(), None).unwrap();
24//! ```
25//!
26//! ## Implementing Custom Processors
27//!
28//! Custom processors can be implemented by creating a new type that implements
29//! the [`Processor`] trait:
30//!
31//! ```rust
32//! # use nucleusflow::core::traits::Processor;
33//! # use nucleusflow::core::error::Result;
34//! # use serde_json::Value;
35//! #[derive(Debug)]
36//! struct CustomProcessor;
37//!
38//! impl Processor for CustomProcessor {
39//!     type Input = String;
40//!     type Output = String;
41//!     type Context = Value;
42//!
43//!     fn process(
44//!         &self,
45//!         input: Self::Input,
46//!         _context: Option<&Self::Context>
47//!     ) -> Result<Self::Output> {
48//!         Ok(input.to_uppercase())
49//!     }
50//! }
51//! ```

63/// Configuration options for content processing.
64#[derive(Debug, Clone, Serialize, Deserialize)]
65pub struct ProcessorConfig {
66    /// Enable content validation
67    #[serde(default = "default_true")]
68    pub validate: bool,
69
70    /// Enable content sanitization
71    #[serde(default = "default_true")]
72    pub sanitize: bool,
73
74    /// Custom processor options
75    #[serde(default)]
76    pub options: HashMap<String, JsonValue>,
77}
78
79impl Default for ProcessorConfig {
80    fn default() -> Self {
81        Self {
82            validate: true,
83            sanitize: true,
84            options: HashMap::new(),
85        }
86    }
87}

89/// Common metadata structure for processed content.
90#[derive(Debug, Clone, Default, Serialize, Deserialize)]
91pub struct ContentMetadata {
92    /// Content title
93    pub title: Option<String>,
94
95    /// Content description
96    pub description: Option<String>,
97
98    /// Publication date
99    pub date: Option<String>,
100
101    /// Content tags
102    pub tags: Vec<String>,
103
104    /// Custom metadata fields
105    pub custom: HashMap<String, JsonValue>,
106}
107
108// Helper function for serde defaults
109fn default_true() -> bool {
110    true
111}
