1//! # Template Rendering Module
2//!
3//! Provides flexible template rendering capabilities with support for various template engines.
4//! The module offers a pluggable architecture for template rendering with built-in support
5//! for Handlebars templates.
6//!
7//! ## Features
8//!
9//! - Pluggable template engine architecture
10//! - Built-in Handlebars support with helpers
11//! - Template caching and validation
12//! - Partial template support
13//! - Custom helper registration
14

27/// Represents a custom template helper with helper name and execution.
28pub trait TemplateHelper: Send + Sync {
29    /// Executes the helper with the given parameters and context.
30    fn execute(
31        &self,
32        params: &[JsonValue],
33        context: &JsonValue,
34    ) -> Result<JsonValue>;
35
36    /// Returns the name of the helper for registration.
37    fn name(&self) -> &str;
38}

40/// Provides details for template validation errors.
41#[derive(Debug, Clone)]
42pub struct ValidationError {
43    /// Error details detailing the validation issue
44    pub details: String,
45    /// Line number where error occurred, if available
46    pub line: Option<usize>,
47    /// Column number where error occurred, if available
48    pub column: Option<usize>,
49    /// Template source snippet where error occurred
50    pub source: Option<String>,
51}
52
53impl From<ValidationError> for ProcessingError {
54    fn from(error: ValidationError) -> Self {
55        ProcessingError::TemplateProcessing {
56            details: error.details,
57            template_name: String::new(),
58            source: None,
59        }
60    }
61}

63/// Renderer for Handlebars templates with caching and custom helpers.
64#[derive(Clone)]
65pub struct HandlebarsRenderer {
66    engine: Arc<RwLock<Handlebars<'static>>>, // Handlebars engine
67    template_dir: PathBuf,                    // Directory for templates
68    template_cache: Arc<RwLock<HashMap<String, String>>>, // Cache for loaded templates
69    helpers: Arc<RwLock<HashMap<String, Box<dyn TemplateHelper>>>>, // Custom registered helpers
70    strict_mode: bool, // Flag for strict mode
71}
72
73impl std::fmt::Debug for HandlebarsRenderer {
74    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
75        f.debug_struct("HandlebarsRenderer")
76            .field("template_dir", &self.template_dir)
77            .field("strict_mode", &self.strict_mode)
78            .finish()
79    }
80}

82impl HandlebarsRenderer {
83    /// Creates a new instance of `HandlebarsRenderer`.
84    pub fn new(template_dir: &Path) -> Result<Self> {
85        let mut handlebars = Handlebars::new();
86        handlebars.set_dev_mode(cfg!(debug_assertions));
87        handlebars.register_escape_fn(handlebars::html_escape);
88
89        let mut renderer = Self {
90            engine: Arc::new(RwLock::new(handlebars)),
91            template_dir: template_dir.to_path_buf(),
92            template_cache: Arc::new(RwLock::new(HashMap::new())),
93            helpers: Arc::new(RwLock::new(HashMap::new())),
94            strict_mode: false,
95        };
96
97        renderer =
98            renderer.with_helper("uppercase", helpers::UppercaseHelper);
99        renderer.load_templates()?;
100        Ok(renderer)
101    }
102
103    /// Enables or disables strict mode, affecting how missing variables and undefined helpers are handled.
104    pub fn with_strict_mode(mut self, strict: bool) -> Self {
105        self.strict_mode = strict;
106        self.engine.write().set_strict_mode(strict);
107        self
108    }
109
110    /// Registers a custom helper with the renderer.
111    pub fn with_helper<H>(self, name: &str, helper: H) -> Self
112    where
113        H: TemplateHelper + Clone + 'static,
114    {
115        _ = self
116            .helpers
117            .write()
118            .insert(name.to_string(), Box::new(helper.clone()));
119        self.register_helper(name, helper);
120        self
121    }
122
123    /// Registers a partial template.
124    pub fn with_partial(
125        self,
126        name: &str,
127        template: &str,
128    ) -> Result<Self> {
129        self.engine
130            .write()
131            .register_partial(name, template)
132            .map_err(|e| ProcessingError::TemplateProcessing {
133                details: format!(
134                    "Failed to register partial '{}': {}",
135                    name, e
136                ),
137                template_name: name.to_string(),
138                source: Some(Box::new(e)),
139            })?;
140        Ok(self)
141    }
142
143    /// Loads templates from the directory, caching and validating them.
144    fn load_templates(&self) -> Result<()> {
145        let mut engine = self.engine.write();
146        let mut cache = self.template_cache.write();
147
148        for entry in
149            std::fs::read_dir(&self.template_dir).map_err(|e| {
150                ProcessingError::TemplateProcessing {
151                    details: format!(
152                        "Failed to read template directory: {}",
153                        e
154                    ),
155                    template_name: String::new(),
156                    source: Some(Box::new(e)),
157                }
158            })?
159        {
160            let entry = entry.map_err(|e| {
161                ProcessingError::TemplateProcessing {
162                    details: format!(
163                        "Failed to read directory entry: {}",
164                        e
165                    ),
166                    template_name: String::new(),
167                    source: Some(Box::new(e)),
168                }
169            })?;
170            let path = entry.path();
171
172            if path.is_file()
173                && path.extension().and_then(|s| s.to_str())
174                    == Some("hbs")
175            {
176                let template_name = path
177                    .file_stem()
178                    .and_then(|s| s.to_str())
179                    .ok_or_else(|| {
180                        ProcessingError::TemplateProcessing {
181                            details: "Invalid template filename"
182                                .to_string(),
183                            template_name: path.display().to_string(),
184                            source: None,
185                        }
186                    })?;
187
188                let template_content = std::fs::read_to_string(&path)
189                    .map_err(|e| {
190                    ProcessingError::TemplateProcessing {
191                        details: format!(
192                            "Failed to read template file: {}",
193                            e
194                        ),
195                        template_name: path.display().to_string(),
196                        source: Some(Box::new(e)),
197                    }
198                })?;
199
200                self.validate_template(&template_content).map_err(
201                    |e| ProcessingError::TemplateProcessing {
202                        details: format!(
203                            "Template validation failed: {}",
204                            e
205                        ),
206                        template_name: template_name.to_string(),
207                        source: None,
208                    },
209                )?;
210
211                engine
212                    .register_template_string(
213                        template_name,
214                        &template_content,
215                    )
216                    .map_err(|e| {
217                        ProcessingError::TemplateProcessing {
218                            details: format!(
219                                "Failed to register template: {}",
220                                e
221                            ),
222                            template_name: template_name.to_string(),
223                            source: Some(Box::new(e)),
224                        }
225                    })?;
226
227                _ = cache.insert(
228                    template_name.to_string(),
229                    template_content,
230                );
231            }
232        }
233        Ok(())
234    }
235
236    /// Registers a helper function with the Handlebars engine.
237    fn register_helper<H>(&self, name: &str, helper: H)
238    where
239        H: TemplateHelper + 'static,
240    {
241        let helper_fn = move |h: &Helper,
242                              _: &Handlebars,
243                              ctx: &Context,
244                              _: &mut RenderContext,
245                              out: &mut dyn Output|
246              -> std::result::Result<
247            (),
248            RenderError,
249        > {
250            let params: Vec<JsonValue> =
251                h.params().iter().map(|p| p.value().clone()).collect();
252
253            let result =
254                helper.execute(&params, ctx.data()).map_err(|e| {
255                    RenderError::from(RenderErrorReason::Other(
256                        e.to_string(),
257                    ))
258                })?;
259            out.write(&result.to_string())?;
260            Ok(())
261        };
262
263        self.engine
264            .write()
265            .register_helper(name, Box::new(helper_fn));
266    }

268    /// Validates the template syntax to catch errors early.
269    fn validate_template(&self, template: &str) -> Result<()> {
270        let engine = self.engine.read();
271        _ = engine
272            .render_template(template, &JsonValue::Null)
273            .map_err(|e| ValidationError {
274                details: e.to_string(),
275                line: e.line_no,
276                column: e.column_no,
277                source: Some(template.to_string()),
278            })?;
279
280        let mut brackets = Vec::new();
281        let mut in_tag = false;
282
283        for (i, c) in template.chars().enumerate() {
284            match c {
285                '{' if in_tag => brackets.push(('{', i)),
286                '}' if brackets.pop().is_none() => {
287                    return Err(ValidationError {
288                        details: "Unmatched closing brace".to_string(),
289                        line: None,
290                        column: Some(i),
291                        source: Some(template.to_string()),
292                    }
293                    .into());
294                }
295                '{' => in_tag = true,
296                _ => {}
297            }
298        }
299
300        if !brackets.is_empty() {
301            return Err(ValidationError {
302                details: "Unmatched opening brace".to_string(),
303                line: None,
304                column: Some(brackets[0].1),
305                source: Some(template.to_string()),
306            }
307            .into());
308        }
309
310        Ok(())
311    }

313    /// Validates template context variables in strict mode.
314    fn validate_context(
315        &self,
316        template: &str,
317        context: &JsonValue,
318    ) -> Result<()> {
319        let template_content = self
320            .template_cache
321            .read()
322            .get(template)
323            .ok_or_else(|| ProcessingError::TemplateProcessing {
324                details: format!(
325                    "Template '{}' not found in cache",
326                    template
327                ),
328                template_name: template.to_string(),
329                source: None,
330            })?
331            .clone();
332
333        let mut required_vars = Vec::new();
334        let mut current_var = String::new();
335        let mut in_var = false;
336
337        for c in template_content.chars() {
338            match c {
339                '{' => {
340                    current_var.clear();
341                    in_var = true;
342                }
343                '}' if in_var => {
344                    required_vars.push(current_var.clone());
345                    in_var = false;
346                }
347                c if in_var => current_var.push(c),
348                _ => {}
349            }
350        }
351
352        for var in required_vars {
353            if context.get(&var).is_none() {
354                return Err(ProcessingError::TemplateProcessing {
355                    details: format!(
356                        "Missing required variable '{}'",
357                        var
358                    ),
359                    template_name: template.to_string(),
360                    source: None,
361                });
362            }
363        }
364
365        Ok(())
366    }
367}

369impl TemplateRenderer for HandlebarsRenderer {
370    fn render(
371        &self,
372        template: &str,
373        context: &JsonValue,
374    ) -> Result<String> {
375        if self.strict_mode {
376            self.validate_context(template, context)?;
377        }
378
379        self.engine.read().render(template, context).map_err(|e| {
380            ProcessingError::TemplateProcessing {
381                details: format!("Template rendering failed: {}", e),
382                template_name: template.to_string(),
383                source: Some(Box::new(e)),
384            }
385        })
386    }
387
388    fn validate(
389        &self,
390        template: &str,
391        context: &JsonValue,
392    ) -> Result<()> {
393        if !self.template_cache.read().contains_key(template) {
394            return Err(ProcessingError::TemplateProcessing {
395                details: format!("Template '{}' not found", template),
396                template_name: template.to_string(),
397                source: None,
398            });
399        }
400
401        if self.strict_mode {
402            self.validate_context(template, context)?;
403        }
404
405        Ok(())
406    }
407}

409/// Built-in helpers for template processing.
410pub mod helpers {
411    use super::*;
412
413    /// Helper to convert text to uppercase.
414    #[derive(Debug, Clone, Copy)]
415    pub struct UppercaseHelper;
416
417    impl TemplateHelper for UppercaseHelper {
418        fn execute(
419            &self,
420            params: &[JsonValue],
421            _context: &JsonValue,
422        ) -> Result<JsonValue> {
423            let text = params
424                .first()
425                .and_then(|p| p.as_str())
426                .ok_or_else(|| ProcessingError::TemplateProcessing {
427                    details:
428                        "Uppercase helper requires a string parameter"
429                            .to_string(),
430                    template_name: String::new(),
431                    source: None,
432                })?;
433            Ok(JsonValue::String(text.to_uppercase()))
434        }
435
436        fn name(&self) -> &str {
437            "uppercase"
438        }
439    }
